## 2. Adversarial Validation (Detecting Train vs. Test Shift)

Before modeling, determine whether the training and test sets come from identical distributions. If a classifier can distinguish between Train and Test rows ($\text{ROC-AUC} \gg 0.50$), feature drift exists:

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import roc_auc_score
import lightgbm as lgb

def run_adversarial_validation(train_df, test_df, ignore_cols=["id", "target"]):
    """
    Labels Train as 0 and Test as 1, trains LightGBM, and returns:
    - Adversarial AUC score (0.50 = identical, >0.70 = severe drift)
    - Top drift-inducing features
    """
    feature_cols = [c for c in train_df.columns if c not in ignore_cols and c in test_df.columns]
    
    df_train = train_df[feature_cols].copy()
    df_test = test_df[feature_cols].copy()
    
    df_train["is_test"] = 0
    df_test["is_test"] = 1
    
    all_df = pd.concat([df_train, df_test], ignore_index=True)
    
    # Encode categoricals
    for c in all_df.select_dtypes(include=["object", "category"]).columns:
        all_df[c] = all_df[c].astype("category").cat.codes
        
    X = all_df[feature_cols].to_numpy()
    y = all_df["is_test"].to_numpy()
    
    skf = StratifiedKFold(n_splits=3, shuffle=True, random_state=42)
    oof = np.zeros(len(y))
    importances = np.zeros(len(feature_cols))
    
    for tr_idx, va_idx in skf.split(X, y):
        clf = lgb.LGBMClassifier(n_estimators=100, learning_rate=0.05, verbose=-1, random_state=42)
        clf.fit(X[tr_idx], y[tr_idx])
        oof[va_idx] = clf.predict_proba(X[va_idx])[:, 1]
        importances += clf.feature_importances_ / 3.0
        
    adv_auc = roc_auc_score(y, oof)
    df_imp = pd.DataFrame({"feature": feature_cols, "importance": importances}).sort_values("importance", ascending=False)
    
    print(f"\n[ADVERSARIAL VALIDATION] AUC: {adv_auc:.4f}")
    if adv_auc > 0.65:
        print("  [ALERT] Substantial distribution drift detected! Top drifting features:")
        print(df_imp.head(5).to_string(index=False))
    else:
        print("  [OK] Train and Test distributions are well-aligned (AUC close to 0.50).")
        
    return adv_auc, df_imp
```

## 3. Synthetic vs. Original Dataset Forensics

Modern tabular benchmarks (e.g. Playground Series) are generated by sampling generative models trained on real-world datasets. Comparing the synthetic dataset to the historical reference exposes the generator's mechanics:

```python
from scipy.spatial import cKDTree
from sklearn.preprocessing import StandardScaler

def audit_synthetic_generator(synthetic_df, original_df, num_cols):
    """
    Analyzes generator perturbation:
    1. Computes distance to nearest original archetype (snap distance distribution).
    2. Computes frequency oversampling/undersampling ratios.
    """
    scaler = StandardScaler()
    orig_scaled = scaler.fit_transform(original_df[num_cols].fillna(0))
    syn_scaled = scaler.transform(synthetic_df[num_cols].fillna(0))
    
    tree = cKDTree(orig_scaled)
    dists, idxs = tree.query(syn_scaled, k=1)
    
    print(f"\n[GENERATOR FORENSICS] Nearest Neighbor Distances:")
    print(f"  Mean Euclidean Distance to Original Archetype: {dists.mean():.4f}")
    print(f"  Median Distance: {np.median(dists):.4f} | Max Distance: {dists.max():.4f}")
    print(f"  Zero Distance (Exact Duplicates of Original): {(dists == 0).sum():,} rows ({(dists == 0).mean()*100:.2f}%)")
    
    # Analyze frequency oversampling on discrete numericals
    for col in num_cols:
        orig_counts = original_df[col].value_counts(normalize=True)
        syn_counts = synthetic_df[col].value_counts(normalize=True)
        
        ratio = (syn_counts / (orig_counts + 1e-6)).dropna()
        if len(ratio) > 0 and (ratio.max() > 3.0 or ratio.min() < 0.3):
            print(f"  Feature '{col}' exhibits significant frequency bias (Max Ratio: {ratio.max():.2f}x).")
            
    return dists, idxs
```

## 4. Bivariate Empirical Target Rate Profiling (Step-Function Discovery)

Tree models construct orthogonal axis-aligned splits that often over-smooth sharp local target rate shifts. Discretizing continuous variables into high-resolution quantiles and computing target rates with Wilson score confidence intervals uncovers non-linear thresholds:

```python
def plot_target_profile(df, col, target_col, n_bins=20):
    """
    Binned target rate analysis to spot non-linearities, plateaus, and step thresholds.
    """
    df = df[[col, target_col]].dropna().copy()
    try:
        df["bin"] = pd.qcut(df[col], q=n_bins, duplicates="drop")
    except ValueError:
        df["bin"] = pd.cut(df[col], bins=n_bins)
        
    stats = df.groupby("bin", observed=True)[target_col].agg(["count", "mean"]).reset_index()
    stats.columns = ["bin_interval", "sample_count", "target_rate"]
    
    # Calculate 95% Wilson confidence interval
    z = 1.96
    p = stats["target_rate"]
    n = stats["sample_count"]
    denominator = 1 + z**2 / n
    centre = (p + z**2 / (2 * n)) / denominator
    spread = z * np.sqrt(p * (1 - p) / n + z**2 / (4 * n**2)) / denominator
    
    stats["ci_lower"] = np.clip(centre - spread, 0, 1)
    stats["ci_upper"] = np.clip(centre + spread, 0, 1)
    
    print(f"\n[BIVARIATE PROFILE] Target Rate across '{col}' Bins:")
    print(stats.head(10).to_string(index=False))
    return stats
```

## 5. Mantissa, Rounding & Benford's Law Forensics

Synthetic generators often output continuous floats where human data entry adheres to round decimals ($0.00, 0.25, 0.50, 0.75, 0.99$). Dissecting decimal mantissas detects generator rounding artifacts:

```python
def audit_decimal_mantissas(df, float_cols, target_col):
    """
    Profiles fractional parts and evaluates correlation with target.
    """
    print("\n[MANTISSA FORENSICS] Evaluating decimal rounding artifacts:")
    for col in float_cols:
        x = df[col].to_numpy()
        frac = x - np.floor(x)
        
        is_exact_zero = (frac < 1e-4).mean()
        is_half = (np.abs(frac - 0.5) < 1e-3).mean()
        is_quarter = (np.min([np.abs(frac - q) for q in [0.25, 0.75]], axis=0) < 1e-3).mean()
        
        print(f"  Feature '{col}':")
        print(f"    Exact Integers: {is_exact_zero*100:.2f}% | Halves: {is_half*100:.2f}% | Quarters: {is_quarter*100:.2f}%")
        
        # Test if round numbers have different target rates
        rate_round = df.loc[frac < 1e-4, target_col].mean()
        rate_noisy = df.loc[frac >= 1e-4, target_col].mean()
        print(f"    Target Rate: Round={rate_round:.4f} vs Noisy={rate_noisy:.4f} (Delta: {abs(rate_round - rate_noisy):.4f})")
```

### Benford's Law Anomaly Screening
```python
def benford_law_divergence(series):
    """
    Measures divergence of leading digits against Benford's logarithmic distribution.
    Divergence = sum((actual - expected)^2 / expected)
    """
    benford_expected = {d: np.log10(1 + 1.0 / d) for d in range(1, 10)}
    digits = series.abs().astype(str).str.replace(r"^0+\.?", "", regex=True).str[0]
    counts = digits.value_counts(normalize=True)
    
    chi_stat = 0.0
    for d in range(1, 10):
        actual = counts.get(str(d), 0.0)
        expected = benford_expected[d]
        chi_stat += ((actual - expected) ** 2) / expected
        
    return chi_stat
```

## 6. Domain Equation Residual Auditing

Whenever domain relationships exist between columns (e.g. $\text{TotalCharges} \approx \text{tenure} \times \text{MonthlyCharges}$, or $\text{BMI} \approx \text{weight} / \text{height}^2$):
1. **Never assume exact equality**: Real business data has mid-cycle changes, penalties, promotional credits, and rounding errors.
2. **Compute the residual**: $e = y_{\text{observed}} - f(X_{\text{domain}})$.
3. **Correlate the residual with target**: A non-zero residual correlation indicates the deviation carries business meaning rather than random noise:

```python
def audit_domain_residual(df, true_col, expected_col, target_col):
    df = df.copy()
    residual = df[true_col] - df[expected_col]
    corr = np.corrcoef(residual, df[target_col])[0, 1]
    print(f"\n[DOMAIN RESIDUAL] '{true_col}' vs '{expected_col}':")
    print(f"  Residual Mean: {residual.mean():.4f} | Std: {residual.std():.4f}")
    print(f"  Correlation with Target: {corr:.4f}")
    if abs(corr) > 0.05:
        print("  [SIGNAL] High residual correlation! Inject residual as a primary feature.")
```

## 7. Categorical Interaction Screening (Information Divergence)

Screening 50+ categorical pairs blindly explodes feature space. Use contingency table divergence to isolate pairs where the joint distribution deviates from independent marginals:

```python
from scipy.stats import chi2_contingency

def screen_categorical_interactions(df, cat_cols, target_col, top_n=5):
    """
    Screens pairwise combinations using Chi-square test of independence on target.
    """
    scores = []
    for i in range(len(cat_cols)):
        for j in range(i + 1, len(cat_cols)):
            c1, c2 = cat_cols[i], cat_cols[j]
            contingency = pd.crosstab(df[c1].astype(str) + "__" + df[c2].astype(str), df[target_col])
            chi2, p, _, _ = chi2_contingency(contingency)
            scores.append((f"{c1}__{c2}", chi2, p))
            
    scores.sort(key=lambda x: x[1], reverse=True)
    print(f"\n[INTERACTION SCREENING] Top {top_n} Categorical Pairs by Target Divergence:")
    for name, chi2, p in scores[:top_n]:
        print(f"  Pair: {name:35s} | Chi2: {chi2:10.1f} | p-value: {p:.2e}")
        
    return [s[0] for s in scores[:top_n]]
```

## 8. Summary: The Grandmaster Forensics Checklist

Before writing any model code, an elite data analyst executes:
